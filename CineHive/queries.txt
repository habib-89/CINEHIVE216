CREATE TABLE ACTOR (
 ACTOR_ID NUMBER GENERATED BY DEFAULT AS IDENTITY,
 ACTOR_NAME VARCHAR2(150) NOT NULL,
 DATE_OF_BIRTH DATE, NATIONALITY VARCHAR2(80),
 PHOTO_URL VARCHAR2(500), BIOGRAPHY VARCHAR2(2000),
 CONSTRAINT PK_ACTOR PRIMARY KEY (ACTOR_ID)
);

CREATE TABLE APP_USER (
 USER_ID NUMBER GENERATED BY DEFAULT AS IDENTITY,
 USERNAME VARCHAR2(80) NOT NULL, EMAIL VARCHAR2(200) NOT NULL,
 PASSWORD_HASH VARCHAR2(255) NOT NULL, DATE_OF_BIRTH DATE,
 PROFILE_PIC VARCHAR2(500), DATE_JOINED DATE NOT NULL,
 CONSTRAINT PK_APP_USER PRIMARY KEY (USER_ID),
 CONSTRAINT UQ_APP_USER_USERNAME UNIQUE (USERNAME),
 CONSTRAINT UQ_APP_USER_EMAIL UNIQUE (EMAIL)
);

CREATE TABLE CINEMA (
 CINEMA_ID NUMBER GENERATED BY DEFAULT AS IDENTITY,
 CINEMA_NAME VARCHAR2(200) NOT NULL, ADDRESS VARCHAR2(300),
 CITY VARCHAR2(100), CONTACT_NUMBER VARCHAR2(30),
 CONSTRAINT PK_CINEMA PRIMARY KEY (CINEMA_ID)
);

CREATE TABLE DIRECTOR (
 DIRECTOR_ID NUMBER GENERATED BY DEFAULT AS IDENTITY,
 DIRECTOR_NAME VARCHAR2(150) NOT NULL, DATE_OF_BIRTH DATE,
 NATIONALITY VARCHAR2(80), PHOTO_URL VARCHAR2(500),
 BIOGRAPHY VARCHAR2(2000),
 CONSTRAINT PK_DIRECTOR PRIMARY KEY (DIRECTOR_ID)
);

CREATE TABLE GENRE (
 GENRE_ID NUMBER GENERATED BY DEFAULT AS IDENTITY,
 GENRE_NAME VARCHAR2(80) NOT NULL,
 CONSTRAINT PK_GENRE PRIMARY KEY (GENRE_ID),
 CONSTRAINT UQ_GENRE_NAME UNIQUE (GENRE_NAME)
);

CREATE TABLE MOVIE (
 MOVIE_ID NUMBER GENERATED BY DEFAULT AS IDENTITY,
 TITLE VARCHAR2(200) NOT NULL, RELEASE_DATE DATE,
 DURATION NUMBER, LANGUAGE VARCHAR2(50),
 DESCRIPTION VARCHAR2(2000), TRAILER_URL VARCHAR2(500),
 POSTER_URL VARCHAR2(500), BOX_OFFICE NUMBER, BUDGET NUMBER,
 CONSTRAINT PK_MOVIE PRIMARY KEY (MOVIE_ID)
);

CREATE TABLE PRODUCTION_COMPANY (
 PRODUCTION_ID NUMBER GENERATED BY DEFAULT AS IDENTITY,
 COMPANY_NAME VARCHAR2(200) NOT NULL, COUNTRY VARCHAR2(100),
 FOUNDED_YEAR NUMBER,
 CONSTRAINT PK_PRODUCTION_COMPANY PRIMARY KEY (PRODUCTION_ID)
);

CREATE TABLE SCREEN (
 SCREEN_ID NUMBER GENERATED BY DEFAULT AS IDENTITY,
 CINEMA_ID NUMBER NOT NULL, SCREEN_NAME VARCHAR2(100) NOT NULL,
 CAPACITY NUMBER NOT NULL,
 CONSTRAINT PK_SCREEN PRIMARY KEY (SCREEN_ID),
 CONSTRAINT FK_SCREEN_CINEMA FOREIGN KEY (CINEMA_ID)
 REFERENCES CINEMA(CINEMA_ID)
);

CREATE TABLE SEAT (
 SEAT_ID NUMBER GENERATED BY DEFAULT AS IDENTITY,
 SCREEN_ID NUMBER NOT NULL, ROW_NUMBER VARCHAR2(10) NOT NULL,
 SEAT_NUMBER NUMBER NOT NULL, SEAT_TYPE VARCHAR2(30) NOT NULL,
 CONSTRAINT PK_SEAT PRIMARY KEY (SEAT_ID),
 CONSTRAINT FK_SEAT_SCREEN FOREIGN KEY (SCREEN_ID)
 REFERENCES SCREEN(SCREEN_ID),
 CONSTRAINT UQ_SEAT_POSITION UNIQUE (SCREEN_ID,ROW_NUMBER,SEAT_NUMBER)
);

CREATE TABLE SHOWTIME (
 SHOWTIME_ID NUMBER GENERATED BY DEFAULT AS IDENTITY,
 MOVIE_ID NUMBER NOT NULL, SCREEN_ID NUMBER NOT NULL,
 SHOW_DATE DATE NOT NULL, START_TIME TIMESTAMP(6) NOT NULL,
 END_TIME TIMESTAMP(6) NOT NULL, TICKET_PRICE NUMBER NOT NULL,
 CONSTRAINT PK_SHOWTIME PRIMARY KEY (SHOWTIME_ID),
 CONSTRAINT FK_SHOWTIME_MOVIE FOREIGN KEY (MOVIE_ID)
 REFERENCES MOVIE(MOVIE_ID),
 CONSTRAINT FK_SHOWTIME_SCREEN FOREIGN KEY (SCREEN_ID)
 REFERENCES SCREEN(SCREEN_ID)
);

CREATE TABLE BOOKING (
 BOOKING_ID NUMBER GENERATED BY DEFAULT AS IDENTITY,
 USER_ID NUMBER NOT NULL, SHOWTIME_ID NUMBER NOT NULL,
 BOOKING_DATE TIMESTAMP(6) NOT NULL, TOTAL_AMOUNT NUMBER NOT NULL,
 PAYMENT_STATUS VARCHAR2(20) NOT NULL,
 CONSTRAINT PK_BOOKING PRIMARY KEY (BOOKING_ID),
 CONSTRAINT FK_BOOKING_USER FOREIGN KEY (USER_ID)
 REFERENCES APP_USER(USER_ID),
 CONSTRAINT FK_BOOKING_SHOWTIME FOREIGN KEY (SHOWTIME_ID)
 REFERENCES SHOWTIME(SHOWTIME_ID),
 CONSTRAINT UQ_BOOKING_ID_SHOWTIME UNIQUE (BOOKING_ID,SHOWTIME_ID)
);

CREATE TABLE BOOKING_SEAT (
 BOOKING_ID NUMBER NOT NULL, SHOWTIME_ID NUMBER NOT NULL,
 SEAT_ID NUMBER NOT NULL,
 CONSTRAINT PK_BOOKING_SEAT PRIMARY KEY (BOOKING_ID,SHOWTIME_ID,SEAT_ID),
 CONSTRAINT FK_BOOKING_SEAT_BOOKING
 FOREIGN KEY (BOOKING_ID,SHOWTIME_ID)
 REFERENCES BOOKING(BOOKING_ID,SHOWTIME_ID),
 CONSTRAINT FK_BOOKING_SEAT_SEAT FOREIGN KEY (SEAT_ID)
 REFERENCES SEAT(SEAT_ID),
 CONSTRAINT UQ_SHOWTIME_SEAT UNIQUE (SHOWTIME_ID,SEAT_ID)
);

CREATE TABLE ACTS_IN (
 ACTOR_ID NUMBER NOT NULL, MOVIE_ID NUMBER NOT NULL, ROLE VARCHAR2(150),
 CONSTRAINT PK_ACTS_IN PRIMARY KEY (ACTOR_ID,MOVIE_ID),
 CONSTRAINT FK_ACTS_IN_ACTOR FOREIGN KEY (ACTOR_ID) REFERENCES ACTOR(ACTOR_ID),
 CONSTRAINT FK_ACTS_IN_MOVIE FOREIGN KEY (MOVIE_ID) REFERENCES MOVIE(MOVIE_ID)
);

CREATE TABLE DIRECTS (
 DIRECTOR_ID NUMBER NOT NULL, MOVIE_ID NUMBER NOT NULL,
 CONSTRAINT PK_DIRECTS PRIMARY KEY (DIRECTOR_ID,MOVIE_ID),
 CONSTRAINT FK_DIRECTS_DIRECTOR FOREIGN KEY (DIRECTOR_ID) REFERENCES DIRECTOR(DIRECTOR_ID),
 CONSTRAINT FK_DIRECTS_MOVIE FOREIGN KEY (MOVIE_ID) REFERENCES MOVIE(MOVIE_ID)
);

CREATE TABLE MOVIE_GENRE (
 MOVIE_ID NUMBER NOT NULL, GENRE_ID NUMBER NOT NULL,
 CONSTRAINT PK_MOVIE_GENRE PRIMARY KEY (MOVIE_ID,GENRE_ID),
 CONSTRAINT FK_MOVIE_GENRE_MOVIE FOREIGN KEY (MOVIE_ID) REFERENCES MOVIE(MOVIE_ID),
 CONSTRAINT FK_MOVIE_GENRE_GENRE FOREIGN KEY (GENRE_ID) REFERENCES GENRE(GENRE_ID)
);

CREATE TABLE PRODUCES (
 PRODUCTION_ID NUMBER NOT NULL, MOVIE_ID NUMBER NOT NULL,
 CONSTRAINT PK_PRODUCES PRIMARY KEY (PRODUCTION_ID,MOVIE_ID),
 CONSTRAINT FK_PRODUCES_COMPANY FOREIGN KEY (PRODUCTION_ID)
 REFERENCES PRODUCTION_COMPANY(PRODUCTION_ID),
 CONSTRAINT FK_PRODUCES_MOVIE FOREIGN KEY (MOVIE_ID) REFERENCES MOVIE(MOVIE_ID)
);

CREATE TABLE RATING (
 USER_ID NUMBER NOT NULL, MOVIE_ID NUMBER NOT NULL,
 RATING_VALUE NUMBER(2,1) NOT NULL, RATING_DATE TIMESTAMP(6) NOT NULL,
 CONSTRAINT PK_RATING PRIMARY KEY (USER_ID,MOVIE_ID),
 CONSTRAINT FK_RATING_USER FOREIGN KEY (USER_ID) REFERENCES APP_USER(USER_ID),
 CONSTRAINT FK_RATING_MOVIE FOREIGN KEY (MOVIE_ID) REFERENCES MOVIE(MOVIE_ID),
 CONSTRAINT CK_RATING_VALUE CHECK (RATING_VALUE>=0 AND RATING_VALUE<=9.9)
);

CREATE TABLE REVIEW (
 REVIEW_ID NUMBER GENERATED BY DEFAULT AS IDENTITY,
 USER_ID NUMBER NOT NULL, MOVIE_ID NUMBER NOT NULL,
 REVIEW_TEXT VARCHAR2(3000) NOT NULL, REVIEW_DATE TIMESTAMP(6) NOT NULL,
 CONSTRAINT PK_REVIEW PRIMARY KEY (REVIEW_ID),
 CONSTRAINT FK_REVIEW_USER FOREIGN KEY (USER_ID) REFERENCES APP_USER(USER_ID),
 CONSTRAINT FK_REVIEW_MOVIE FOREIGN KEY (MOVIE_ID) REFERENCES MOVIE(MOVIE_ID)
);

CREATE TABLE WATCHLIST (
 USER_ID NUMBER NOT NULL, MOVIE_ID NUMBER NOT NULL,
 ADDED_DATE TIMESTAMP(6) NOT NULL,
 CONSTRAINT PK_WATCHLIST PRIMARY KEY (USER_ID,MOVIE_ID),
 CONSTRAINT FK_WATCHLIST_USER FOREIGN KEY (USER_ID) REFERENCES APP_USER(USER_ID),
 CONSTRAINT FK_WATCHLIST_MOVIE FOREIGN KEY (MOVIE_ID) REFERENCES MOVIE(MOVIE_ID)
);

INSERT INTO SEAT (SCREEN_ID, ROW_NUMBER, SEAT_NUMBER, SEAT_TYPE)
SELECT
  s.SCREEN_ID,
  CHR(64 + r.ROW_NO),
  n.SEAT_NO,
  CASE WHEN r.ROW_NO <= 2 THEN 'PREMIUM' ELSE 'REGULAR' END
FROM SCREEN s
CROSS JOIN (SELECT LEVEL AS ROW_NO FROM dual CONNECT BY LEVEL <= 5) r
CROSS JOIN (SELECT LEVEL AS SEAT_NO FROM dual CONNECT BY LEVEL <= 8) n
WHERE s.SCREEN_ID IN (1, 2);
